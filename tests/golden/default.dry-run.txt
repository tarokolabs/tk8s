# ---- <TMP>/clusters/tk8s/router/scripts/vyos-preconfig-bootup.script
#!/bin/sh
# Generated by tkctl for cluster tk8s. VyOS runs this before it loads config.boot. tkctl
# renders tkctl.boot; a config.boot without VyOS's own version footer would be migrated at
# every boot (blocks added, the file rewritten), so the footer is appended here, taken from the
# default config of the VyOS build that is about to load it.
cp /opt/vyatta/etc/config/tkctl.boot /opt/vyatta/etc/config/config.boot
grep '^// ' /opt/vyatta/etc/config.boot.default >> /opt/vyatta/etc/config/config.boot

# ---- <TMP>/clusters/tk8s/router/scripts/vyos-postconfig-bootup.script
#!/bin/sh
# Generated by tkctl for cluster tk8s. podman gives every container on the cluster network a
# default route to the router; this container is the router, and that route would shadow the
# static default to the host (VyOS's zebra ranks a kernel route above its own static one).
ip route del default via 172.22.0.253 2>/dev/null || true
# When the container stops, netavark deletes exactly that route (gateway, metric 100, proto
# static) and fails the unit when it is gone. A transient unit puts it back at shutdown; by then
# nothing routes through here any more.
# The host bridge sits on the same link as the nodes, so for every packet it forwards the
# router would send the node an ICMP redirect to the host (ping shows them as errors, and a
# node that accepted them would bypass the router). VyOS has no CLI knob for this.
sysctl -qw net.ipv4.conf.all.send_redirects=0 net.ipv4.conf.eth0.send_redirects=0
systemd-run --quiet --unit=tk8s-podman-route -p RemainAfterExit=yes -p "ExecStop=/usr/sbin/ip route replace default via 172.22.0.253 metric 100 proto static" /bin/true

[Unit]
Description=tk8s cluster tk8s router (VyOS at 172.22.0.253: the nodes' gateway and the apiserver load balancer)
PartOf=tk8s.target

[Container]
ContainerName=tk8s-router
Image=ghcr.io/tarokolabs/tk8s/router:2026.10.07-0712
# The container's hostname must be the one config.boot sets: VyOS runs sudo at every commit and a
# hostname it cannot resolve stalls the boot for minutes.

IP=172.22.0.253
# tkctl.boot and the bootup scripts, rendered from cluster.yaml; VyOS writes config.boot and its archive here.

PodmanArgs=--privileged --systemd=always
# systemd's shutdown signal; the image sets it too.
StopSignal=SIGRTMIN+3
StopTimeout=30

[Service]
# VyOS's systemd exits 130 on SIGRTMIN+3, like the node image's.
SuccessExitStatus=130
Restart=always

[Install]
WantedBy=tk8s.target

[Unit]
Description=tk8s node tk8s-control-plane-1 (control-plane) of cluster tk8s

[Container]

HostName=tk8s-control-plane-1
Network=tk8s.network
IP=172.22.0.1
Environment=TZ=Asia/Taipei
# api.<cluster>.k8s is the controlPlaneEndpoint: every node resolves it to the router (proxy
# mode, the kubeadm HA picture: one load balancer in front of the control planes) or to the
# anycast VIP the control planes announce to the router (bgp mode).
AddHost=api.tk8s.k8s:172.22.0.253
Volume=<TMP>/cni:/opt/cni/bin
Volume=/lib/modules:/lib/modules:ro
# Quadlet recreates the container on every start (--rm --replace). Everything the node
# must keep across restarts and reboots lives in named volumes; podman copies the image
# content into an empty volume on first use.
Volume=tk8s-control-plane-1-var:/var:suid,exec,dev,rbind
Volume=tk8s-control-plane-1-etc:/etc
Volume=tk8s-control-plane-1-usr-local-bin:/usr/local/bin
# The host's shared image store for this Kubernetes version (images:ensure fills it). CRI-O
# reads it as an additional image store, so nothing a create needs is pulled; read-only
# because one writer per store is the rule. Podman mounts it after the /var volume (deeper path).
Volume=<TMP>/images/1.37.1:/var/lib/containers/shared:ro
Volume=<TMP>/clusters/tk8s:/opt/taroko
Volume=<TMP>/wulin:/opt/taroko/wulin
StopTimeout=60
PodmanArgs=--privileged --cgroupns=private --cpus 2 --memory 4g

[Service]
# The node image stops with SIGRTMIN+3; systemd inside exits 130 on halt.
SuccessExitStatus=130 143
Restart=on-failure
TimeoutStartSec=300

[Install]
WantedBy=multi-user.target
