$ORIGIN tk8s.k8s.
$TTL 10
@                    IN SOA   dns.tk8s.k8s. root.tk8s.k8s. ( 1 10 10 60 10 )
@                    IN NS    dns.tk8s.k8s.
dns                  IN A     172.22.0.253

tk8s-worker2         IN A     172.22.0.3
_apiserver._tcp      IN SRV   0 100 6443 tk8s-control-plane.tk8s.k8s.

# ---- <TMP>/clusters/tk8s/lb/haproxy.cfg
# Generated by tkctl for cluster tk8s: the host-side apiserver load balancer at 172.22.0.100.
# TCP passthrough to every control plane found through the SRV record; health checks over TLS.
global
    log stdout format raw local0
    stats socket /tmp/haproxy.sock mode 600 level admin

defaults
    mode tcp
    log global
    option tcplog
    timeout connect 5s
    timeout client 10m
    timeout server 10m

resolvers dns
    parse-resolv-conf
    resolve_retries 3
    timeout resolve 2s
    timeout retry 2s
    hold valid 10s
    hold obsolete 120s
    hold nx 120s
    hold refused 120s
    hold timeout 120s
    hold other 120s

frontend apiserver
    bind *:6443
    default_backend control-planes

frontend stats
    mode http
    bind 127.0.0.1:8404
    stats enable
    stats uri /stats

backend control-planes
    balance roundrobin
    option httpchk GET /readyz
    http-check expect status 200
    default-server check inter 2s fall 3 rise 2 check-ssl verify none init-addr none resolvers dns resolve-prefer ipv4
    server-template cp 5 _apiserver._tcp.tk8s.k8s

# Generated by tkctl for cluster tk8s (control-plane). Edit cluster.yaml and re-run tkctl instead.
[Unit]
Description=tk8s node tk8s-control-plane (control-plane) of cluster tk8s
PartOf=tk8s.target
After=tk8s-dns.service

[Container]
ContainerName=tk8s-control-plane
Image=ghcr.io/tarokolabs/tk8s/node:v1.37.0
HostName=tk8s-control-plane
Network=tk8s.network
IP=172.22.0.1
Environment=TZ=Asia/Taipei
# The cluster DNS (apiserver SRV records) is this container's upstream resolver; api.<cluster>.k8s
# is the controlPlaneEndpoint: a worker reaches it through its own HAProxy on 127.0.0.1, a control
# plane through the host-side LB (its own apiserver is not up yet when it joins).
DNS=172.22.0.253
AddHost=api.tk8s.k8s:172.22.0.100
Volume=<TMP>/cni:/opt/cni/bin
Volume=/lib/modules:/lib/modules:ro
# Quadlet recreates the container on every start (--rm --replace). Everything the node
# must keep across restarts and reboots lives in named volumes; podman copies the image
# content into an empty volume on first use.
Volume=tk8s-control-plane-var:/var:suid,exec,dev,rbind
Volume=tk8s-control-plane-etc:/etc
Volume=tk8s-control-plane-usr-local-bin:/usr/local/bin
# The host's shared image store for this Kubernetes version (images:ensure fills it). CRI-O
# reads it as an additional image store, so nothing a create needs is pulled; read-only
# because one writer per store is the rule. Podman mounts it after the /var volume (deeper path).
Volume=<TMP>/images/1.37.0:/var/lib/containers/shared:ro
Volume=<TMP>/clusters/tk8s:/opt/taroko
Volume=<TMP>/wulin:/opt/taroko/wulin
StopTimeout=60
PodmanArgs=--privileged --cgroupns=private --cpus 2 --memory 4g

[Service]
# The node image stops with SIGRTMIN+3; systemd inside exits 130 on halt.
SuccessExitStatus=130 143
Restart=on-failure
TimeoutStartSec=300

[Install]
WantedBy=multi-user.target
