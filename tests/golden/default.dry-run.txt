# ---- <TMP>/clusters/tk8s/lb/haproxy.cfg
# Generated by tkctl for cluster tk8s: the host-side apiserver load balancer at 172.22.0.100.
# TCP passthrough to every control plane in cluster.yaml (re-rendered and reloaded by add/delete node);
# health checks over TLS; a control plane marked DOWN loses its established connections so clients
# reconnect elsewhere at once.
global
    log stdout format raw local0
    stats socket /tmp/haproxy.sock mode 600 level admin

defaults
    mode tcp
    log global
    option tcplog
    timeout connect 5s
    timeout client 10m
    timeout server 10m

frontend apiserver
    bind *:6443
    default_backend control-planes

frontend stats
    mode http
    bind 127.0.0.1:8404
    stats enable
    stats uri /stats

backend control-planes
    balance roundrobin
    option httpchk GET /readyz
    http-check expect status 200
    default-server check inter 2s fall 3 rise 2 on-marked-down shutdown-sessions check-ssl verify none
    server tk8s-control-plane-1 172.22.0.1:6443

# Generated by tkctl for cluster tk8s. Edit cluster.yaml and re-run tkctl instead.
[Unit]
Description=tk8s cluster tk8s apiserver load balancer (HAProxy at 172.22.0.100)
PartOf=tk8s.target

[Container]
ContainerName=tk8s-lb
Image=docker.io/library/haproxy:3.3.16-trixie
Network=tk8s.network
IP=172.22.0.100
# The directory, not the file: haproxy.cfg is replaced by rename when a control plane is added or
# removed, and a bind mount of one file would keep showing the inode the container started with.
Volume=<TMP>/clusters/tk8s/lb:/usr/local/etc/haproxy:ro
# The image's stop signal is SIGUSR1 (graceful: keep serving established connections for up to
# timeout client), so a cluster stop would end in podman's SIGKILL and a failed unit.
StopSignal=SIGTERM

[Service]
Restart=always

[Install]
WantedBy=tk8s.target

# api.<cluster>.k8s is the controlPlaneEndpoint and resolves to the host-side LB for every node

Volume=/lib/modules:/lib/modules:ro
# Quadlet recreates the container on every start (--rm --replace). Everything the node
# must keep across restarts and reboots lives in named volumes; podman copies the image
# content into an empty volume on first use.
Volume=tk8s-control-plane-1-var:/var:suid,exec,dev,rbind
Volume=tk8s-control-plane-1-etc:/etc
Volume=tk8s-control-plane-1-usr-local-bin:/usr/local/bin
# The host's shared image store for this Kubernetes version (images:ensure fills it). CRI-O
# reads it as an additional image store, so nothing a create needs is pulled; read-only
# because one writer per store is the rule. Podman mounts it after the /var volume (deeper path).
Volume=<TMP>/images/1.37.1:/var/lib/containers/shared:ro
Volume=<TMP>/clusters/tk8s:/opt/taroko
Volume=<TMP>/wulin:/opt/taroko/wulin
StopTimeout=60
PodmanArgs=--privileged --cgroupns=private --cpus 2 --memory 4g

[Install]
WantedBy=multi-user.target
